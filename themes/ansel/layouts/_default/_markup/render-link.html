{{- $link := strings.Replace .Destination "_index.md" "" -}} {{- $isRemote := strings.HasPrefix $link "http" -}} {{- $isAnchor := strings.HasPrefix $link "#" -}} {{/* A shortcode in the destination -- [text]({{< relref "..." >}}) -- reaches the hook as Hugo's own placeholder, the real URL being spliced in once the page is rendered. Resolving it here would consume the placeholder and leave the link pointing nowhere. */}} {{- $isShortcode := strings.HasPrefix $link "HAHAHUGOSHORTCODE" -}} {{- $isWikipedia := strings.Contains $link "wikipedia.org" -}} {{- $isNetwork := strings.Contains $link "ansel.photos" -}} {{- $isDownload := eq (lower .Title) "download" -}} {{- $title := .Title -}} {{- $page := .Page -}} {{- if and $isWikipedia (not $title) -}} {{- $title = "Wikipedia link" -}} {{- end -}} {{- if and $isRemote (and (not $isNetwork) (not $title)) -}} {{- $title = "External link" -}} {{- end -}} {{- if and (not $isShortcode) (and (not $isRemote) (not $isAnchor)) -}} {{/* For local links, we want writers to use relative links to (mostly .md) files, as much as possible. These links can be followed in modern text editors on the file system, without building the website. But Hugo needs links to be relative to the generated website structure, and the conversion is not straight-forward, especially with translations. */}} {{- $url := urls.Parse $link -}} {{- if $url.Path -}} {{- $fragment := printf "%s" $url.Fragment -}} {{- $link = $url.Path -}} {{- $output := "" -}} {{- if strings.HasSuffix $link ".xml" -}} {{- $output = "rss" -}} {{- $link = strings.Replace $link ".xml" "" -}} {{- else if strings.HasSuffix $link ".md" -}} {{- $output = "html" -}} {{- $link = strings.Replace $link ".md" "" -}} {{- end -}} {{/* A leaf bundle is addressed by its directory: "index" is the file a writer follows in an editor, but it is not part of the page's path. */}} {{- $link = strings.TrimSuffix "/index" $link -}} {{- $got_it := false -}} {{/* Not every link target is a page. A file sitting next to the page in its bundle -- an XMP sidecar, a spreadsheet -- is a page resource, and is served as-is. */}} {{- with $page.Resources.GetMatch $link -}} {{- $link = .RelPermalink -}} {{- $got_it = true -}} {{- end -}} {{/* This works as-is only for relative links to pages from the same section/branch bundle */}} {{- if not $got_it -}} {{- with $page.GetPage $link -}} {{- $link = .RelPermalink -}} {{- $got_it = true -}} {{- end -}} {{- end -}} {{- if not $got_it -}} {{- with site.GetPage $link -}} {{- $link = .RelPermalink -}} {{- $got_it = true -}} {{- end -}} {{- end -}} {{/* Resolve the link the way the writer wrote it: as a path on disk, relative to the file the link is in. path.Join folds the "../" away, which is what neither GetPage nor `ref` does from inside a page bundle -- there, "../" has to climb out of the bundle directory, and both read it as climbing out of the section instead. */}} {{- if not $got_it -}} {{- with $page.File -}} {{- with site.GetPage (path.Join "/" (path.Dir .Path) $link) -}} {{- $link = .RelPermalink -}} {{- $got_it = true -}} {{- end -}} {{- end -}} {{- end -}} {{/* This works only for relative links pointing to pages outside the current section/branch bundle and if the current page is NOT IN a page boundle. see https://gohugo.io/content-management/page-bundles/ */}} {{- if not $got_it -}} {{- $opts := dict "path" $link "lang" .Page.Lang "outputFormat" $output -}} {{- with try (ref .Page $opts) -}} {{- with $page.GetPage .Value -}} {{- $link = .RelPermalink -}} {{- $got_it = true -}} {{- end -}} {{- end -}} {{- end -}} {{/* This works only for relative links pointing to pages outside the current section/branch bundle and if the current page is IN a page boundle. see https://gohugo.io/content-management/page-bundles/ */}} {{- if not $got_it -}} {{- $opts := dict "path" (strings.TrimPrefix "../" $link) "lang" .Page.Lang "outputFormat" $output -}} {{- with try (ref .Page $opts) -}} {{- with $page.GetPage .Value -}} {{- $link = .RelPermalink -}} {{- $got_it = true -}} {{- end -}} {{- end -}} {{- end -}} {{- if not $got_it -}} {{- errorf "Internal link %q (%q) NOT
